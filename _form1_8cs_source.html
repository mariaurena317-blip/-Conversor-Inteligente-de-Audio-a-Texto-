233 private void ProcesarArchivoBinario(string ruta)
234 {
235 try
236 {
237 FileInfo info = new FileInfo(ruta);
238
239
240 string metaInfo = $"Nombre: {info.Name}\r\n" +
241 $"Formato: {info.Extension}\r\n" +
242 $"Tamaño (KB): {(info.Length / 1024.0):F2}\r\n" +
243 $"Creación: {info.CreationTime}";
244
245
246 byte[] bytes = new byte[600];
247 using (FileStream fs = new FileStream(ruta, FileMode.Open, FileAccess.Read))
248 {
249
250 fs.Read(bytes, 0, bytes.Length);
251 }
252
253
254 StringBuilder hexBuilder = new StringBuilder();
255 for (int i = 0; i < bytes.Length; i++)
256 {
257
258 hexBuilder.AppendFormat("{0:X2} ", bytes[i]);
259
260 if ((i + 1) % 16 == 0)
261 {
262 hexBuilder.AppendLine();
263 }
264 }
265
266 string contenidoComoTexto = hexBuilder.ToString().Trim();
267
268
269 txtInfoArchivo.Text = metaInfo + "\r\n\r\n=== Formato de Almacenamiento (Código Máquina - HEX) ===\r\n\r\n" + contenidoComoTexto;
270 }
271 catch (Exception ex)
272 {
273
274 MessageBox.Show("Error al leer el archivo: " + ex.Message, "Error de Lectura", MessageBoxButtons.OK, MessageBoxIcon.Error);
275 }
